cmake_minimum_required(VERSION 3.21)
project(unified_ui_preview C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This preview uses the Windows desktop display backend")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LVGL_SOURCE_DIR "${ROOT}/managed_components/lvgl__lvgl" CACHE PATH "LVGL 9.5 source tree")
set(LV_BUILD_CONF_PATH "${ROOT}/src/platforms/goldieos/components/ui/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
if(NOT DEFINED UI_PREVIEW_SOURCE)
  set(UI_PREVIEW_SOURCE "${ROOT}/src/application/ui/ai_lvgl.c")
endif()
if(NOT DEFINED UI_PREVIEW_TARGET)
  set(UI_PREVIEW_TARGET unified_ui_preview)
endif()
add_executable(${UI_PREVIEW_TARGET} WIN32 main.c
  "${UI_PREVIEW_SOURCE}"
  "${ROOT}/src/application/core/ai_app.c")
target_include_directories(${UI_PREVIEW_TARGET} PRIVATE
  "${ROOT}/src/application/ui/include" "${ROOT}/src/application/core/include")
target_link_libraries(${UI_PREVIEW_TARGET} PRIVATE lvgl gdi32 user32)
if(DEFINED UI_PREVIEW_WINDOW_TITLE)
  target_compile_definitions(${UI_PREVIEW_TARGET} PRIVATE UI_PREVIEW_WINDOW_TITLE="${UI_PREVIEW_WINDOW_TITLE}")
endif()
if(MINGW)
  target_link_options(${UI_PREVIEW_TARGET} PRIVATE -static)
endif()
